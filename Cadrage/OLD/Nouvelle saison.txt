-- Ordre important : on supprime d'abord ce qui dépend des matchs/séries,
-- avant de pouvoir toucher aux matchs et séries eux-mêmes (contraintes FK).

-- re-déterminées par la simulation au fil de la nouvelle saison).

set team1_id = null, team2_id = null, official_status = 'SCHEDULED',
    official_winner_team_id = null, official_score_format = null,
    next_series_slot = next_series_slot -- (inchangé, juste pour lisibilité)
